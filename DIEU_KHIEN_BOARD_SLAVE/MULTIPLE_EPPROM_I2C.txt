// Đọc data từ EEPROM256 khi gặp mã nhận diện STEP, trả về giá trị số thứ tự step trong chương trình
uint16_t Read_Program_STEP (EEPROM_25LC256 &EEPROM256, uint16_t Address){
  uint8_t data_index_step[3];
  uint8_t result = 1;
  result = EEPROM256.Read_Mutiple_data_from_EEPROM (Address, 3);
  if (result == 0){ 
    for (int i = 0; i < 3; i++) { 
      data_index_step[i] = EEPROM256.getResponseBuffer(i); 
    } 
  }
  if (data_index_step[0] == STEP) { return UInt16_word(data_index_step, 1); }
  else return 0xFFFF;
  }

void Write_to_EEPROM(EEPROM_24C256 &EEPROM256, uint16_t Address, uint16_t data){
  uint8_t trans_data[2];
  uint8_t result;
  trans_data[0] = highByte(data); //Serial.print("highByte: "); Serial.print(trans_data[0],HEX);
  trans_data[1] = lowByte(data); //Serial.print("lowByte: "); Serial.print(trans_data[1],HEX);
  result = EEPROM256.Write_Mutiple_data_to_EEPROM(trans_data, 0x0000, 2);
}
